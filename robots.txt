# Zollstock soll nicht in Suchergebnissen auftauchen.
#
# Dafür steht in jeder Seite ein <meta name="robots" content="noindex">.
# Damit eine Suchmaschine das lesen kann, muss sie die Seiten auch abrufen
# dürfen – ein pauschales "Disallow: /" würde genau das verhindern und die
# nackte Adresse trotzdem im Index landen lassen. Suchmaschinen dürfen also
# herkommen und bekommen dann das noindex zu sehen.

User-agent: *
Allow: /
Disallow: /tests/
Disallow: /docs/
